字段2:skill_count 技能总数
sql逻辑
SELECT COUNT(DISTINCT s.skill_id) AS skill_count
FROM swe_cron_executions e
JOIN swe_cron_jobs j
  ON e.job_id = j.id
JOIN (
    SELECT
        source_id,
        skill_id,
        MIN(cn_name) AS cn_name,
        MIN(skill_name) AS skill_name
    FROM swe_marketplace_skills
    WHERE include_in_statistics = 1
      AND skill_id IS NOT NULL
      AND skill_id != ''
    GROUP BY source_id, skill_id
) s
  ON FIND_IN_SET(s.skill_id, j.skill_ids)
 AND s.source_id = j.source_id
WHERE j.deleted_at IS NULL
  AND j.status <> 'deleted'
